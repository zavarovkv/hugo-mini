{{- /* Goldmark supplies smart punctuation and --/---. This pass adds conservative Russian nesting, prose dashes and spacing; it never guesses numeric ranges or mathematical operators. */ -}} {{- $text := string .text -}} {{- $lang := lower (index (split .lang "-") 0) -}} {{- $quoteStack := .quoteStack -}} {{- if in (slice "ru" "en") $lang -}} {{- $space := `[ \t\x{00a0}\x{202f}]` -}} {{- $units := `₽|€|\$|°C|°F|кг|мг|г|км|см|мм|м²|м³|м|мл|л|мс|мин|ч|с|КБ|МБ|ГБ|ТБ|кВт|Вт|Гц|МГц|kg|mg|g|km|cm|mm|m²|m³|m|ml|ms|min|KB|MB|GB|TB|kB|kW|W|Hz|MHz|px` -}} {{- $parts := slice -}} {{- /* Keep visible URLs byte-for-byte, including percent escapes. */ -}} {{- range findRESubmatch `(?is)(.*?)([A-Za-z][A-Za-z0-9+.-]*://[^\s<]+|mailto:[^\s<]+|tel:[^\s<]+|www\.[^\s<]+|/[^\s<]+|$)` $text -}} {{- $prose := index . 1 -}} {{- $url := index . 2 -}} {{- /* Existing digit groups only: years and unformatted IDs are not regrouped. */ -}} {{- $numbers := slice -}} {{- $groups := printf `(?s)(.*?)([\p{L}\p{N}_]+(?:%s[0-9]{3})+(?:[.,][0-9]+)?[\p{L}\p{N}_]*|$)` $space -}} {{- range findRESubmatch $groups $prose -}} {{- $number := index . 2 -}} {{- if findRE (printf `^[0-9]{1,3}(?:%s[0-9]{3})+(?:[.,][0-9]+)?$` $space) $number -}} {{- $number = replaceRE $space "\u202f" $number -}} {{- end -}} {{- $numbers = $numbers | append (index . 1) $number -}} {{- end -}} {{- $prose = delimit $numbers "" -}} {{- /* Match whole candidates, including word prefixes/suffixes, before validating. This leaves model16px and 10MBps intact and handles adjacent measurements without consuming their shared boundary. */ -}} {{- $measurements := slice -}} {{- $candidates := printf `(?s)(.*?)([\p{L}\p{N}_]*[0-9][\p{L}\p{N}_]*(?:[.,][0-9]+)?%s*(?:%s|[%%‰])[\p{L}\p{N}_²³]*|$)` $space $units -}} {{- range findRESubmatch $candidates $prose -}} {{- $measurement := index . 2 -}} {{- $measurement = replaceRE (printf `^([0-9]+(?:[.,][0-9]+)?)%s*(%s)$` $space $units) "$1\u00a0$2" $measurement -}} {{- $percentSpace := "" -}} {{- if eq $lang "ru" -}}{{- $percentSpace = "\u00a0" -}}{{- end -}} {{- $measurement = replaceRE (printf `^([0-9]+(?:[.,][0-9]+)?)%s*([%%‰])$` $space) (printf "$1%s$2" $percentSpace) $measurement -}} {{- $measurements = $measurements | append (index . 1) $measurement -}} {{- end -}} {{- $prose = delimit $measurements "" -}} {{- if eq $lang "ru" -}} {{- /* Bind only common compound prepositions with a nonbreaking hyphen. */ -}} {{- $prose = replaceRE `(?i)(^|[^\p{L}\p{N}_])(из)-(за|под)` "$1$2\u2011$3" $prose -}} {{- /* Keep conventional Russian abbreviations on one line. */ -}} {{- $prose = replaceRE `(?i)(^|[^\p{L}\p{N}_])(т)\.[ \t\x{00a0}]*(е|д|п|к|н|о)\.` "$1$2.\u00a0$3." $prose -}} {{- /* A short conjunction plus preposition can form one small group. */ -}} {{- $short := `в|к|с|у|о|и|а|на|по|из|от|до|за|во|со|ко|об` -}} {{- $chain := printf `(?i)(^|[ \t\x{00a0}\x{202f}\(\[«„]|«|“|„)(%s)[ \t]+(%s)[ \t]+([\p{L}]{3,12})` $short $short -}} {{- $prose = replaceRE $chain "$1$2\u00a0$3\u00a0$4" $prose -}} {{- /* Only a spaced, word-like hyphen becomes an em dash. Arithmetic, ranges and compound words are intentionally left alone. */ -}} {{- $prose = replaceRE `([\p{L}\)\]»]|»|”)[ \t]+-[ \t]+([\p{L}])` "$1 — $2" $prose -}} {{- $prose = replaceRE `(\S)[ \t\x{00a0}\x{202f}]+(—|—|—|—)[ \t\x{00a0}\x{202f}]+` "$1\u00a0$2 " $prose -}} {{- /* Bind short Russian function words to a following word, without creating long unbreakable chains or touching punctuation/code. */ -}} {{- $prose = replaceRE (printf `(?i)(^|[ \t\x{00a0}\x{202f}\(\[«„]|«|“|„)(%s)[ \t]+([\p{L}]{3,})` $short) "$1$2\u00a0$3" $prose -}} {{- /* Enclitic particles stay with the preceding word. */ -}} {{- $prose = replaceRE `(?i)([\p{L}\p{N}])[ \t]+(же|ли|ль|бы|б)([ \t.,!?;:…»”\)\]]|»|”|$)` "$1\u00a0$2$3" $prose -}} {{- $quoted := partial "typography/quotes.html" (dict "text" $prose "quoteStack" $quoteStack) -}} {{- $prose = $quoted.text -}} {{- $quoteStack = $quoted.quoteStack -}} {{- /* A URL can end a prose quotation. Consume only trailing closing marks when a quote is already open; keep the URL itself verbatim. */ -}} {{- if gt (len $quoteStack) 0 -}} {{- with findRESubmatch `^(.*?)((?:»|”|“|’|[»”“’])+[.,!?;:]*)$` $url -}} {{- $match := index . 0 -}} {{- $ending := partial "typography/quotes.html" (dict "text" (index $match 2) "quoteStack" $quoteStack) -}} {{- $url = printf "%s%s" (index $match 1) $ending.text -}} {{- $quoteStack = $ending.quoteStack -}} {{- end -}} {{- end -}} {{- end -}} {{- $parts = $parts | append $prose $url -}} {{- end -}} {{- $text = delimit $parts "" -}} {{- end -}} {{- return (dict "text" $text "quoteStack" $quoteStack) -}}
