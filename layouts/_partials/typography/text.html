{{- /* Goldmark supplies smart punctuation and --/---. This pass adds conservative Russian nesting, prose dashes and spacing; it never guesses numeric ranges or mathematical operators. */ -}} {{- $text := string .text -}} {{- $lang := lower (index (split .lang "-") 0) -}} {{- $quoteStack := .quoteStack -}} {{- if in (slice "ru" "en") $lang -}} {{- $space := `[ \t\x{00a0}\x{202f}]` -}} {{- $parts := slice -}} {{- /* Keep visible URLs byte-for-byte, including percent escapes. */ -}} {{- range findRESubmatch `(?is)(.*?)([A-Za-z][A-Za-z0-9+.-]*://[^\s<]+|mailto:[^\s<]+|tel:[^\s<]+|www\.[^\s<]+|/[^\s<]+|$)` $text -}} {{- $prose := index . 1 -}} {{- $url := index . 2 -}} {{- if findRE `[0-9]` $prose 1 -}} {{- $prose = partial "typography/numbers.html" (dict "text" $prose "lang" $lang "space" $space) -}} {{- end -}} {{- if eq $lang "ru" -}} {{- /* Bind only common compound prepositions with a nonbreaking hyphen. */ -}} {{- $prose = replaceRE `(?i)(^|[^\p{L}\p{N}_])(из)-(за|под)` "$1$2\u2011$3" $prose -}} {{- /* Keep conventional Russian abbreviations on one line. */ -}} {{- $prose = replaceRE `(?i)(^|[^\p{L}\p{N}_])(т)\.[ \t\x{00a0}]*(е|д|п|к|н|о)\.` "$1$2.\u00a0$3." $prose -}} {{- /* A short conjunction plus preposition can form one small group. */ -}} {{- $short := `в|к|с|у|о|и|а|на|по|из|от|до|за|во|со|ко|об` -}} {{- $chain := printf `(?i)(^|[ \t\x{00a0}\x{202f}\(\[«„]|«|“|„)(%s)[ \t]+(%s)[ \t]+([\p{L}]{3,12})` $short $short -}} {{- $prose = replaceRE $chain "$1$2\u00a0$3\u00a0$4" $prose -}} {{- /* Only a spaced, word-like hyphen becomes an em dash. Arithmetic, ranges and compound words are intentionally left alone. */ -}} {{- $prose = replaceRE `([\p{L}\)\]»]|»|”)[ \t]+-[ \t]+([\p{L}])` "$1 — $2" $prose -}} {{- $prose = replaceRE `(\S)[ \t\x{00a0}\x{202f}]+(—|—|—|—)[ \t\x{00a0}\x{202f}]+` "$1\u00a0$2 " $prose -}} {{- /* Bind short Russian function words to a following word, without creating long unbreakable chains or touching punctuation/code. */ -}} {{- $prose = replaceRE (printf `(?i)(^|[ \t\x{00a0}\x{202f}\(\[«„]|«|“|„)(%s)[ \t]+([\p{L}]{3,})` $short) "$1$2\u00a0$3" $prose -}} {{- /* Enclitic particles stay with the preceding word. */ -}} {{- $prose = replaceRE `(?i)([\p{L}\p{N}])[ \t]+(же|ли|ль|бы|б)([ \t.,!?;:…»”\)\]]|»|”|$)` "$1\u00a0$2$3" $prose -}} {{- if findRE `[&«»„“”‘’]` $prose 1 -}} {{- $quoted := partial "typography/quotes.html" (dict "text" $prose "quoteStack" $quoteStack) -}} {{- $prose = $quoted.text -}} {{- $quoteStack = $quoted.quoteStack -}} {{- end -}} {{- /* A URL can end a prose quotation. Consume only trailing closing marks when a quote is already open; keep the URL itself verbatim. */ -}} {{- if gt (len $quoteStack) 0 -}} {{- with findRESubmatch `^(.*?)((?:»|”|“|’|[»”“’])+[.,!?;:]*)$` $url -}} {{- $match := index . 0 -}} {{- $ending := partial "typography/quotes.html" (dict "text" (index $match 2) "quoteStack" $quoteStack) -}} {{- $url = printf "%s%s" (index $match 1) $ending.text -}} {{- $quoteStack = $ending.quoteStack -}} {{- end -}} {{- end -}} {{- end -}} {{- $parts = $parts | append $prose $url -}} {{- end -}} {{- $text = delimit $parts "" -}} {{- end -}} {{- return (dict "text" $text "quoteStack" $quoteStack) -}}
