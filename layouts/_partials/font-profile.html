{{- /* Site-owned font assets; this theme never includes a consumer's font files. Callers cache by language and useDefaultFonts, and key CSS by this result. */ -}} {{- $profile := dict "custom" false "regular" 300 "medium" 500 "bold" 600 "title" 500 -}} {{- $config := .Site.Params.fonts -}} {{- if and $config (not .Params.useDefaultFonts) -}} {{- if not (and $config.family (findRE `^[\pL\pN _-]+$` $config.family)) -}} {{- errorf "fonts.family must be a plain font family name" -}} {{- end -}} {{- $titleFamily := $config.family -}} {{- with $config.title -}} {{- if not (and .family (findRE `^[\pL\pN _-]+$` .family)) -}} {{- errorf "fonts.title.family must be a plain font family name" -}} {{- end -}} {{- if eq .family $config.family -}}{{ errorf "fonts.title.family must differ from fonts.family; use fonts.faces for the same family" }}{{- end -}} {{- $titleFamily = .family -}} {{- end -}} {{- $weights := dict "regular" (int ($config.regularWeight | default 400)) "medium" (int ($config.mediumWeight | default 400)) "bold" (int ($config.boldWeight | default 700)) "title" (int ($config.titleWeight | default 700)) -}} {{- range $role, $weight := $weights -}} {{- if or (lt $weight 1) (gt $weight 1000) -}}{{ errorf "fonts: invalid %s weight %d" $role $weight }}{{- end -}} {{- end -}} {{- $faces := slice -}} {{- $normalWeights := slice -}} {{- $boldItalic := false -}} {{- range $config.faces -}} {{- $style := .style | default "normal" -}} {{- $weight := int .weight -}} {{- $faces = $faces | append (partial "font-face.html" (dict "family" $config.family "source" .source "style" $style "weight" $weight "preload" (.preload | default false))) -}} {{- if eq $style "normal" -}}{{ $normalWeights = $normalWeights | append $weight }}{{- end -}} {{- if and (eq $style "italic") (eq $weight $weights.bold) -}}{{ $boldItalic = true }}{{- end -}} {{- end -}} {{- with $config.title -}} {{- $faces = $faces | append (partial "font-face.html" (dict "family" $titleFamily "source" .source "weight" $weights.title "preload" (.preload | default false))) -}} {{- end -}} {{- range $role, $weight := $weights -}} {{- if and (not (and (eq $role "title") $config.title)) (not (in $normalWeights $weight)) -}}{{ errorf "fonts: %s weight %d has no normal face" $role $weight }}{{- end -}} {{- end -}} {{- $profile = merge $weights (dict "custom" true "family" $config.family "titleFamily" $titleFamily "faces" $faces "boldItalic" $boldItalic) -}} {{- end -}} {{- return $profile -}}
