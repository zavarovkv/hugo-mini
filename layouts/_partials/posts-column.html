{{- /* Renders one column of category groups for the blog listing. Inputs (from posts-column dict): .groups — slice of category slugs (or "_untagged" for posts with no category) .allPosts — posts, already filtered for `hidden` and sorted newest first Within each category, posts with `pinned = true` are floated to the top (preserving reverse-chronological order between themselves), the rest follow in reverse-chronological order. */ -}} {{- $groups := .groups -}} {{- $allPosts := .allPosts -}} {{- $newCutoff := now.AddDate 0 0 (int (mul -1 (site.Params.newPostDays | default 30))) -}}
{{- range $groups }} {{- $tag := . -}} {{- $title := "" -}} {{- $catPosts := slice -}} {{- if eq $tag "_untagged" -}} {{- $title = i18n "misc" -}} {{- $catPosts = where $allPosts ".Params.categories" nil -}} {{- else -}} {{- $title = i18n (printf "cat_%s" $tag) | default $tag -}} {{- $catPosts = where $allPosts ".Params.categories" "intersect" (slice $tag) -}} {{- end -}} {{- $pinned := where $catPosts ".Params.pinned" true -}} {{- $unpinned := where $catPosts ".Params.pinned" "ne" true -}} {{- $ordered := $pinned | append $unpinned -}}

{{ $title }}

    {{- range $ordered }}
  • {{ .Title }}{{ if ge .Date $newCutoff }} {{ i18n "new_badge" | default "New" }}{{ end }}
    • {{- end }}
{{- end }}
