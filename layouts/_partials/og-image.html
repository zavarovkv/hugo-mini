{{/* OG Image Generator - creates unique 1200x630 images */}} {{- $base := resources.Get "og/og-base.png" -}} {{- $boldFont := resources.Get "fonts/Inter-Bold.ttf" -}} {{- $regularFont := resources.Get "fonts/Inter-Regular.ttf" -}} {{- if and $base $boldFont $regularFont -}} {{- $title := "" -}} {{- $subtitle := "" -}} {{- $path := "" -}} {{- /* params.author is a table (it carries jobTitle), so it must not be used as a display-name fallback: that renders the map itself into the image as "map[jobtitle:…]". Site title is the fallback instead. */ -}} {{- $authorName := .Site.Params.authorName | default .Site.Title -}} {{- $attribution := printf "%s · %s" (.Site.BaseURL | replaceRE "^https?://" "" | replaceRE "/$" "") $authorName -}} {{- if .IsPage -}} {{/* Blog post: title + attribution */}} {{- $title = .Title -}} {{- $subtitle = $attribution -}} {{- $path = printf "%sog.png" .RelPermalink -}} {{- else if .IsHome -}} {{/* Homepage */}} {{- $title = $authorName -}} {{- $subtitle = .Site.Params.description | default .Site.Title -}} {{- $path = printf "%sog.png" .RelPermalink -}} {{- else if .IsSection -}} {{/* Section pages (e.g. /blog/) */}} {{- $title = (.Title | default .Site.Title) -}} {{- $subtitle = $attribution -}} {{- $path = printf "%sog.png" .RelPermalink -}} {{- end -}} {{- if $title -}} {{- $img := $base.Filter (images.Text $title (dict "color" "#ffffff" "size" 52 "linespacing" 8 "x" 80 "y" 80 "font" $boldFont )) -}} {{- $img = $img.Filter (images.Text $subtitle (dict "color" "#888888" "size" 24 "x" 80 "y" 540 "font" $regularFont )) -}} {{- $img = resources.Copy $path $img -}} {{- end -}} {{- else -}} {{/* Fallback: use static default image if it exists */}} {{- if fileExists "static/images/og-default.png" -}} {{- end -}} {{- end -}}
