{{- /* Resolve internal page links before selecting an available translation. Page lookup leaves static assets and external destinations alone. */ -}} {{- $destination := .Destination -}} {{- $url := urls.Parse $destination -}} {{- $page := .Page -}} {{- $base := urls.Parse $page.Site.BaseURL -}} {{- $internal := and (or (not $url.Scheme) (in (slice "http" "https") $url.Scheme)) (or (not $url.Host) (eq $url.Host $base.Host)) -}} {{- if and $internal $url.Path ($page.Site.Params.localizeLinks | default true) -}} {{- /* Explicit /ru/ (or another non-default language prefix) is a deliberate language choice; only unqualified links should follow the page language. */ -}} {{- $explicitLanguage := false -}} {{- range (partialCached "sites.html" $page) -}} {{- $languageRoot := .Home.RelPermalink -}} {{- if and (ne $languageRoot $base.Path) (hasPrefix $url.Path $languageRoot) -}} {{- $explicitLanguage = true -}} {{- end -}} {{- end -}} {{- /* GetPage resolves "." against the content tree. In a browser "./#id" refers to the published article itself, even when its slug is custom. */ -}} {{- $pageURL := urls.Parse $page.Permalink -}} {{- $resolved := $pageURL.ResolveReference $url -}} {{- $target := $page -}} {{- if ne $resolved.Path $pageURL.Path -}} {{- $target = $page.GetPage $url.Path -}} {{- end -}} {{- if not $target -}} {{- $wanted := $url.Path -}} {{- if not (hasPrefix $wanted "/") -}} {{- $directory := $page.RelPermalink -}} {{- if not (hasSuffix $directory "/") -}}{{- $directory = path.Dir $directory -}}{{- end -}} {{- $wanted = path.Join $directory $wanted -}} {{- end -}} {{- $index := partialCached "page-url-index.html" $page -}} {{- $target = index $index (strings.TrimSuffix "/" $wanted) -}} {{- end -}} {{- with $target -}} {{- if not $explicitLanguage -}} {{- range .AllTranslations -}} {{- if eq .Language.Lang $page.Language.Lang -}}{{- $target = . -}}{{- end -}} {{- end -}} {{- end -}} {{- $destination = $target.RelPermalink -}} {{- with $url.RawQuery -}}{{- $destination = printf "%s?%s" $destination . -}}{{- end -}} {{- /* Keep the original escaped fragment, including Unicode anchors. */ -}} {{- with findRE `#.*$` $.Destination 1 -}} {{- $destination = print $destination (index . 0) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $destination -}}
