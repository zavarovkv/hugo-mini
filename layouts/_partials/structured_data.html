{{- /* Schema.org JSON-LD. Built with dict + jsonify rather than hand-written JSON: the previous version concatenated strings with conditional commas, which is one forgotten comma away from emitting invalid JSON that no build step would catch. Emits, depending on page kind: post — BlogPosting + BreadcrumbList page — WebPage + BreadcrumbList home — WebSite + Person other — WebSite */ -}} {{- /* Author identity, shared by every shape below. params.author is a table (it carries jobTitle), so authorName is the display name and the table is only ever read through `index`, which tolerates it being unset. */ -}} {{- $authorName := site.Params.authorName | default site.Title -}} {{- $author := dict "@type" "Person" "name" $authorName -}} {{- with site.Params.authorURL -}} {{- $author = merge $author (dict "url" .) -}} {{- end -}} {{- $blocks := slice -}} {{- if .IsPage -}} {{- $description := .Description | default .Summary -}} {{- /* og.png is produced by og-image.html only when the base image and both fonts resolve as resources — assert the image under the same condition so JSON-LD never points at a file that was not generated. */ -}} {{- $hasOgImage := and (resources.Get "og/og-base.png") (resources.Get "fonts/Inter-Bold.ttf") (resources.Get "fonts/Inter-Regular.ttf") -}} {{- $isPost := in (site.Params.mainSections | default (slice "blog")) .Type -}} {{- if $isPost -}} {{- $post := dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" .Title "description" $description "author" $author "publisher" (dict "@type" "Person" "name" $authorName) "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) -}} {{- with .Params.categories -}} {{- $post = merge $post (dict "keywords" (delimit . ", ")) -}} {{- end -}} {{- if $hasOgImage -}} {{- $post = merge $post (dict "image" (dict "@type" "ImageObject" "url" (printf "%sog.png" .Permalink) "width" 1200 "height" 630 )) -}} {{- end -}} {{- $blocks = $blocks | append $post -}} {{- else -}} {{- $blocks = $blocks | append (dict "@context" "https://schema.org" "@type" "WebPage" "name" .Title "description" $description "url" .Permalink "author" $author ) -}} {{- end -}} {{- /* Breadcrumbs: Home › [section] › page. The section step is included only for pages that live in a section with its own listing page. */ -}} {{- $crumbs := slice (dict "@type" "ListItem" "position" 1 "name" (i18n "home") "item" site.Home.Permalink) -}} {{- $sectionPage := .CurrentSection -}} {{- if and $sectionPage $sectionPage.Title (ne $sectionPage.Permalink site.Home.Permalink) -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" (add (len $crumbs) 1) "name" ($sectionPage.Title | default (i18n "articles")) "item" $sectionPage.Permalink ) -}} {{- end -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" (add (len $crumbs) 1) "name" .Title "item" .Permalink ) -}} {{- $blocks = $blocks | append (dict "@context" "https://schema.org" "@type" "BreadcrumbList" "itemListElement" $crumbs ) -}} {{- else -}} {{- $site := dict "@context" "https://schema.org" "@type" "WebSite" "name" site.Title "url" site.BaseURL "author" $author -}} {{- with site.Params.description -}} {{- $site = merge $site (dict "description" .) -}} {{- end -}} {{- $blocks = $blocks | append $site -}} {{- if .IsHome -}} {{- $person := dict "@context" "https://schema.org" "@type" "Person" "name" $authorName "url" site.BaseURL -}} {{- with index (site.Params.author | default dict) "jobTitle" -}} {{- $person = merge $person (dict "jobTitle" .) -}} {{- end -}} {{- $sameAs := slice -}} {{- range $key, $url := site.Params.social -}} {{- if and $url (ne $key "email") -}} {{- $sameAs = $sameAs | append $url -}} {{- end -}} {{- end -}} {{- with $sameAs -}} {{- $person = merge $person (dict "sameAs" .) -}} {{- end -}} {{- $blocks = $blocks | append $person -}} {{- end -}} {{- end -}} {{- range $blocks }} {{- end }}
