{{- /* Validate and fingerprint one site-owned font without changing its bytes. */ -}} {{- $style := .style | default "normal" -}} {{- $weight := int .weight -}} {{- $extension := lower (path.Ext .source) -}} {{- $format := index (dict ".woff2" "woff2" ".woff" "woff" ".otf" "opentype" ".ttf" "truetype") $extension -}} {{- $mime := index (dict ".woff2" "font/woff2" ".woff" "font/woff" ".otf" "font/otf" ".ttf" "font/ttf") $extension -}} {{- if or (not $format) (not (in (slice "normal" "italic") $style)) (lt $weight 1) (gt $weight 1000) -}} {{- errorf "fonts: invalid face %q (use WOFF2/WOFF/OTF/TTF, normal/italic and weight 1–1000)" .source -}} {{- end -}} {{- $face := dict -}} {{- with resources.Get .source -}} {{- $asset := . | resources.Fingerprint "sha256" -}} {{- $face = dict "family" $.family "url" $asset.RelPermalink "style" $style "weight" $weight "format" $format "mime" $mime "preload" ($.preload | default false) -}} {{- else -}} {{- errorf "fonts: missing site asset %q; supply licensed files through a private build source" .source -}} {{- end -}} {{- return $face -}}
