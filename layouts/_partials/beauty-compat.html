{{- /* Legacy public classes are a theme adapter, never an upstream CSS fork. Match complete selector tokens, not prefixes such as .beauty-caption-extra. The input is the pinned, generated Beauty artifact, not arbitrary CSS. */ -}} {{- $css := . -}} {{- range $selector, $alias := dict ".beauty-emphasis" ".underline" ".beauty-caption" ".caption" ".beauty-button" ".button" "blockquote.beauty-callout" "q" -}} {{- $pattern := printf `([(+~,> ])%s([),:+~> .#\[]|$)` (replace $selector "." `\.`) -}} {{- $css = replaceRE $pattern (printf `${1}:is(%s,%s)${2}` $selector $alias) $css -}} {{- end -}} {{- return $css -}}
