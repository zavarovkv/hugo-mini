{{- /* Meta tags: primary, Open Graph, Twitter Card. Title and description are resolved once at the top and reused, so the plain meta, og: and twitter: variants can never drift apart — they previously repeated the expression, and only the plain one fell back to .Summary, which left og:description empty on any page without an explicit description. Schema.org lives in structured_data.html (JSON-LD). Hugo's internal _internal/schema.html is deliberately not used: it emits itemprop attributes that require an enclosing itemscope/itemtype, which this theme's markup does not provide, so the microdata was inert while still duplicating everything JSON-LD already states. */ -}} {{- $title := .Title | default site.Title -}} {{- $description := "" -}} {{- if .Description -}} {{- $description = .Description -}} {{- else if .IsPage -}} {{- $description = .Summary -}} {{- else -}} {{- $description = site.Params.description -}} {{- end -}} {{- $description = $description | plainify | truncate 300 -}} {{- /* OG locale comes from each language's own configuration (see og-locale.html) instead of a hardcoded en/ru pair, so a third language is not mislabelled as Russian. */ -}} {{- $ogLocale := partial "og-locale.html" site.Language -}} {{- if .IsPage }} {{- with .Params.categories }} {{- end }} {{- end }} {{- with $ogLocale }} {{- end }} {{- range .Translations }} {{- with partial "og-locale.html" .Language }} {{- end }} {{- end }} {{- partial "og-image.html" . -}}
